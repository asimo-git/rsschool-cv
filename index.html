*** 5 kyu - Function Cache ***

function cache(func) {
  const cached = {};

  return (...args) => {
    const key = JSON.stringify(args);
    if (cached.hasOwnProperty(key)) {
      return cached[key];
    }
    const res = func(...args);
    cached[key] = res;
    return res;
  };
}
